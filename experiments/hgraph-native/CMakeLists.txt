cmake_minimum_required(VERSION 3.25)
project(builder_comparison LANGUAGES CXX)

# A separately built provider is shared by both development and release consumers.
if(BENCH_RUNTIME)
    add_library(bench_runtime SHARED IMPORTED)
    set_target_properties(bench_runtime PROPERTIES IMPORTED_LOCATION "${BENCH_RUNTIME}")
else()
    find_package(hgraph CONFIG REQUIRED)
    add_library(bench_runtime SHARED runtime.cpp bridge.cpp)
    target_link_libraries(bench_runtime PRIVATE hgraph::core)
endif()
target_compile_features(bench_runtime INTERFACE cxx_std_23)
target_include_directories(bench_runtime INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT BENCH_RUNTIME)
    target_compile_features(bench_runtime PRIVATE cxx_std_23)
endif()
add_library(cpp_nodes STATIC cpp/nodes.cpp)
target_link_libraries(cpp_nodes PUBLIC bench_runtime)
add_executable(cpp_case cpp/app.cpp)
target_link_libraries(cpp_case PRIVATE cpp_nodes)
